    14→// ─── Phone Mask Helper ─────────────────────────────
    15→function formatPhoneMask(value: string): string {
    16→  const digits = value.replace(/\D/g, '').slice(0, 10)
    17→  if (digits.length === 0) return ''
    18→  if (digits.length <= 3) return `+7 ${digits}`
    19→  if (digits.length <= 6) return `+7 ${digits.slice(0, 3)} ${digits.slice(3)}`
    20→  if (digits.length <= 8) return `+7 ${digits.slice(0, 3)} ${digits.slice(3, 6)}-${digits.slice(6)}`
    21→  return `+7 ${digits.slice(0, 3)} ${digits.slice(3, 6)}-${digits.slice(6, 8)}-${digits.slice(8)}`
    22→}

    24→// ─── Floating Label Input ─────────────────────────────
    25→function FloatingInput({
    26→  label,
    27→  value,
    28→  onChange,
    29→  placeholder,
    30→  type = 'text',
    31→  readOnly = false,
    32→  rightIcon = false,
    33→  mask,
    34→}: {
    35→  label: string
    36→  value?: string
    37→  onChange?: (val: string) => void
    38→  placeholder?: string
    39→  type?: string
    40→  readOnly?: boolean
    41→  rightIcon?: boolean
    42→  mask?: 'phone'
    43→}) {
    44→  const hasValue = !!value
    45→  const pr = rightIcon ? 'pr-10' : 'pr-4'
    46→
    47→  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    48→    if (!onChange || readOnly) return
    49→    let raw = e.target.value
    50→    if (mask === 'phone') {
    51→      // Store only digits (strip +7 prefix if present)
    52→      const digits = raw.replace(/\D/g, '')
    53→      // If starts with 7 or 8, skip country code digit
    54→      if (digits.length > 0 && (digits[0] === '7' || digits[0] === '8')) {
    55→        raw = digits.slice(1)
    56→      } else {
    57→        raw = digits
    58→      }
    59→    }
    60→    onChange(raw)
    61→  }
    62→
    63→  const displayValue = mask === 'phone' && value ? formatPhoneMask(value) : (value || '')
    64→
    65→  return (
    66→    <div className="relative">
    67→      <input
    68→        type={type}
    69→        value={displayValue}
    70→        onChange={handleChange}
    71→        placeholder={hasValue ? ' ' : (placeholder || label)}
    72→        className={`w-full h-[48px] px-4 ${pr} ${hasValue ? 'pt-[12px]' : ''} bg-gray-100 rounded-lg text-sm text-gray-900 border-[1.5px] border-transparent focus:border-gray-900 focus:bg-white focus:outline-none transition-colors placeholder:text-gray-400 ${readOnly ? 'cursor-default' : ''}`}
    73→        readOnly={readOnly}
    74→      />
    75→      {hasValue && (
    76→        <span className="absolute left-4 top-[8px] text-[10px] text-gray-500 font-medium pointer-events-none leading-none">
    77→          {label}
    78→        </span>
    79→      )}
    80→    </div>
    81→  )
    82→}
