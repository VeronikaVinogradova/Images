    10→// ─── Phone Mask Helper ─────────────────────────────
    11→function formatPhoneMask(value: string): string {
    12→  const digits = value.replace(/\D/g, '').slice(0, 10)
    13→  if (digits.length === 0) return ''
    14→  if (digits.length <= 3) return `+7 ${digits}`
    15→  if (digits.length <= 6) return `+7 ${digits.slice(0,3)} ${digits.slice(3)}`
    16→  if (digits.length <= 8) return `+7 ${digits.slice(0,3)} ${digits.slice(3,6)}-${digits.slice(6)}`
    17→  return `+7 ${digits.slice(0,3)} ${digits.slice(3,6)}-${digits.slice(6,8)}-${digits.slice(8)}`
    18→}

    20→// ─── Floating Label Input ─────────────────────────────
    21→function FloatingInput({ 
    22→  label, 
    23→  value, 
    24→  onChange,
    25→  placeholder, 
    26→  type = 'text',
    27→  readOnly = false,
    28→  rightIcon = false,
    29→  mask,
    30→}: { 
    31→  label: string
    32→  value?: string
    33→  onChange?: (val: string) => void
    34→  placeholder?: string
    35→  type?: string
    36→  readOnly?: boolean
    37→  rightIcon?: boolean
    38→  mask?: 'phone'
    39→}) {
    40→  const hasValue = !!value
    41→  const pr = rightIcon ? 'pr-10' : 'pr-4'
    42→  
    43→  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    44→    if (!onChange || readOnly) return
    45→    let raw = e.target.value
    46→    if (mask === 'phone') {
    47→      raw = formatPhoneMask(raw)
    48→    }
    49→    onChange(raw)
    50→  }
    51→
    52→  const displayValue = mask === 'phone' && value ? formatPhoneMask(value) : (value || '')
    53→  
    54→  return (
    55→    <div className="relative">
    56→      <input
    57→        type={type}
    58→        value={displayValue}
    59→        onChange={handleChange}
    60→        placeholder={hasValue ? ' ' : (placeholder || label)}
    61→        className={`w-full h-[48px] px-4 ${pr} ${hasValue ? 'pt-[12px]' : ''} bg-gray-100 rounded-lg text-sm text-gray-900 border-[1.5px] border-transparent focus:border-gray-900 focus:bg-white focus:outline-none transition-colors placeholder:text-gray-400 ${readOnly ? 'cursor-default' : ''}`}
    62→        readOnly={readOnly}
    63→      />
    64→      {hasValue && (
    65→        <span className="absolute left-4 top-[8px] text-[10px] text-gray-500 font-medium pointer-events-none leading-none">
    66→          {label}
    67→        </span>
    68→      )}
    69→    </div>
    70→  )
    71→}
