    10→// ─── Floating Label Input ─────────────────────────────
    11→function FloatingInput({ 
    12→  label, 
    13→  value, 
    14→  placeholder, 
    15→  type = 'text',
    16→  readOnly = false,
    17→  rightIcon = false,
    18→}: { 
    19→  label: string
    20→  value?: string
    21→  placeholder?: string
    22→  type?: string
    23→  readOnly?: boolean
    24→  rightIcon?: boolean
    25→}) {
    26→  const hasValue = !!value
    27→  const pr = rightIcon ? 'pr-10' : 'pr-4'
    28→  
    29→  return (
    30→    <div className="relative">
    31→      <input
    32→        type={type}
    33→        defaultValue={value}
    34→        placeholder={hasValue ? ' ' : (placeholder || label)}
    35→        className={`w-full h-[48px] px-4 ${pr} ${hasValue ? 'pt-[12px]' : ''} bg-gray-100 rounded-lg text-sm text-gray-900 border border-transparent focus:border-blue-500 focus:bg-white focus:outline-none transition-colors placeholder:text-gray-400 ${readOnly ? 'cursor-default' : ''}`}

    37→      />
    38→      {hasValue && (
    39→        <span className="absolute left-4 top-[8px] text-[10px] text-gray-500 font-medium pointer-events-none leading-none">
    40→          {label}
    41→        </span>

    45→}
