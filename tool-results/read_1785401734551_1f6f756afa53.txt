    12→// ─── Phone Mask Helper ─────────────────────────────
    13→function formatPhoneMask(value: string): string {
    14→  const digits = value.replace(/\D/g, '').slice(0, 10)
    15→  if (digits.length === 0) return ''
    16→  if (digits.length <= 3) return `+7 ${digits}`
    17→  if (digits.length <= 6) return `+7 ${digits.slice(0, 3)} ${digits.slice(3)}`
    18→  if (digits.length <= 8) return `+7 ${digits.slice(0, 3)} ${digits.slice(3, 6)}-${digits.slice(6)}`
    19→  return `+7 ${digits.slice(0, 3)} ${digits.slice(3, 6)}-${digits.slice(6, 8)}-${digits.slice(8)}`
    20→}

    22→// ─── Floating Label Input ─────────────────────────────
    23→function FloatingInput({
    24→  label,
    25→  value,
    26→  onChange,
    27→  placeholder,
    28→  type = 'text',
    29→  readOnly = false,
    30→  rightIcon = false,
    31→  mask,
    32→}: {
    33→  label: string
    34→  value?: string
    35→  onChange?: (val: string) => void
    36→  placeholder?: string
    37→  type?: string
    38→  readOnly?: boolean
    39→  rightIcon?: boolean
    40→  mask?: 'phone'
    41→}) {
    42→  const hasValue = !!value
    43→  const pr = rightIcon ? 'pr-10' : 'pr-4'
    44→
    45→  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    46→    if (!onChange || readOnly) return
    47→    let raw = e.target.value
    48→    if (mask === 'phone') {
    49→      // Store only digits (strip +7 prefix if present)
    50→      const digits = raw.replace(/\D/g, '')
    51→      // If starts with 7 or 8, skip country code digit
    52→      if (digits.length > 0 && (digits[0] === '7' || digits[0] === '8')) {
    53→        raw = digits.slice(1)
    54→      } else {
    55→        raw = digits
    56→      }
    57→    }
    58→    onChange(raw)
    59→  }
    60→
    61→  const displayValue = mask === 'phone' && value ? formatPhoneMask(value) : (value || '')
    62→
    63→  return (
    64→    <div className="relative">
    65→      <input
    66→        type={type}
    67→        value={displayValue}
    68→        onChange={handleChange}
    69→        placeholder={hasValue ? ' ' : (placeholder || label)}
    70→        className={`w-full h-[48px] px-4 ${pr} ${hasValue ? 'pt-[12px]' : ''} bg-gray-100 rounded-lg text-sm text-gray-900 border-[1.5px] border-transparent focus:border-gray-900 focus:bg-white focus:outline-none transition-colors placeholder:text-gray-400 ${readOnly ? 'cursor-default' : ''}`}
    71→        readOnly={readOnly}
    72→      />
    73→      {hasValue && (
    74→        <span className="absolute left-4 top-[8px] text-[10px] text-gray-500 font-medium pointer-events-none leading-none">
    75→          {label}
    76→        </span>
    77→      )}
    78→    </div>
    79→  )
    80→}
